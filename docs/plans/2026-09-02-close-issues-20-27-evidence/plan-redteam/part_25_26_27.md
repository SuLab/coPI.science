def _auth(user_id) -> dict:
    signer = TimestampSigner(get_settings().secret_key)
    data = base64.b64encode(json.dumps({"user_id": str(user_id)}).encode())
    return {"Cookie": f"copi-session={signer.sign(data).decode()}"}


class _EmptyResult:
    def __iter__(self):
        return iter(())

    def scalar(self):
        return None

    def scalar_one_or_none(self):
        return None


class _RecordingSession:
    def __init__(self, statements):
        self._statements = statements

    async def __aenter__(self):
        return self

    async def __aexit__(self, *exc):
        return False

    async def execute(self, stmt, *args, **kwargs):
        self._statements.append(" ".join(str(stmt).split()))
        return _EmptyResult()

def app_and_statements(monkeypatch):

    return create_app(), statements


def _badge_queries(statements: list[str]) -> list[str]:
    return [s for s in statements if any(t in s for t in _BADGE_TABLES)]


async def test_static_path_runs_no_badge_queries(app_and_statements):
    app, statements = app_and_statements
    async with httpx.AsyncClient(transport=ASGITransport(app=app), base_url="http://t") as c:

### B12 — 27.5: the staleness diff can never match

`pip-compile` writes a header recording the exact command, including `--output-file=<path>`. The
committed lock says `requirements.lock`; the scratch regeneration says `/tmp/tmp.XXXX`. So
`diff -q "$LOCK_TMP" requirements.lock` always differs and `ci.sh` always exits 1 at that step.
(Not runnable here — pip-tools is absent and installing it needs network; confidence: high, from
pip-tools' documented header behaviour.)

**Fix — comment-insensitive comparison (tool-version independent):**
```bash
echo "==> lockfile freshness (requirements.lock matches pyproject.toml)"
LOCK_TMP="$(mktemp)"
if ! "$VENV_PY" -m piptools compile --generate-hashes --no-header \
      --output-file "$LOCK_TMP" pyproject.toml >/dev/null 2>&1; then
  echo "ERROR: pip-compile failed to resolve pyproject.toml — see above." >&2
  rm -f "$LOCK_TMP"; exit 1
fi
# Compare pins only: pip-compile's header records the --output-file path it was
# given, so a raw diff against a temp file can never match even when the pins
# are identical.
if ! diff -q <(grep -v '^#' requirements.lock) <(grep -v '^#' "$LOCK_TMP") >/dev/null; then
  echo "ERROR: requirements.lock is stale — pyproject.toml changed without regenerating it." >&2
  echo "Run: $VENV_PY -m piptools compile --generate-hashes --no-header -o requirements.lock pyproject.toml" >&2
  diff <(grep -v '^#' requirements.lock) <(grep -v '^#' "$LOCK_TMP") >&2 || true
  rm -f "$LOCK_TMP"; exit 1
fi
rm -f "$LOCK_TMP"
echo "    requirements.lock is current"
```
Add `--no-header` to 27.4's generation command too, so the committed file and the check agree.
Also change the prerequisite probe — `python -m piptools --version` is not a documented invocation:
```bash

```
